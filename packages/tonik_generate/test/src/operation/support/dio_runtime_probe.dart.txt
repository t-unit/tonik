import 'dart:convert';
import 'dart:typed_data';

import 'package:dio/dio.dart';
import 'package:dio_runtime_api/src/operation/dio_operation.dart';
import 'package:dio_runtime_api/src/operation/multipart_request.dart';

import 'package:dio_runtime_api/src/operation/void_response.dart';
import 'package:dio_runtime_api/src/model/upload_body.dart';
import 'package:tonik_util/tonik_util.dart';

Future<void> main(List<String> arguments) async {
  final scenario = arguments.single;
  switch (scenario) {
    case 'encoding':
      final stack = StackTrace.fromString('dio encoding stack');
      final error = _expectError(
        await _DioEncodingOperation().run(stack),
        TonikErrorType.encoding,
        response: false,
      );
      _check(error.stackTrace.toString() == stack.toString(), 'encoding stack');
    case 'uri-encoding':
      _expectError(
        await ValueResponse('http://[', _unusedDio).call(),
        TonikErrorType.encoding,
        response: false,
      );
    case 'pre-cancel':
      final cancellation = TonikCancellation()..cancel('before');
      final result = await ValueResponse(
        'https://example.test',
        _successDio,
      ).call(cancellation: cancellation);
      final error = _expectError(
        result,
        TonikErrorType.cancelled,
        response: false,
      );
      _check(error.stackTrace.toString().isNotEmpty, 'missing cancel stack');
      _check(
        error.stackTrace.toString() ==
            (error.error as DioException).stackTrace.toString(),
        'pre-cancel stack was not taken from the cancellation exception',
      );
    case 'accessor':
      final stack = StackTrace.fromString('dio accessor stack');
      Dio accessor() => Error.throwWithStackTrace(StateError('closed'), stack);
      final error = _expectError(
        await ValueResponse('https://example.test', accessor).call(),
        TonikErrorType.other,
        response: false,
      );
      _check(error.stackTrace.toString() == stack.toString(), 'accessor stack');
    case 'in-flight-cancel':
      final cancellation = TonikCancellation();
      final dio = Dio()..httpClientAdapter = _CancelAdapter(cancellation);
      final future = ValueResponse(
        'https://example.test',
        () => dio,
      ).call(cancellation: cancellation);
      final error = _expectError(
        await future,
        TonikErrorType.cancelled,
        response: false,
      );
      _check(error.error is DioException, 'cancel error type');
    case 'cancel-response':
      final native = _nativeResponse();
      final dio = Dio()
        ..httpClientAdapter = _ThrowAdapter(
          DioException(
            requestOptions: native.requestOptions,
            type: DioExceptionType.cancel,
            response: native,
          ),
        );
      final error = _expectError(
        await ValueResponse('https://example.test', () => dio).call(),
        TonikErrorType.cancelled,
        response: true,
      );
      _check(identical(error.response, native), 'cancel response identity');
    case 'transport':
      final native = _nativeResponse();
      final stack = StackTrace.fromString('dio transport stack');
      final dio = Dio()
        ..httpClientAdapter = _ThrowAdapter(
          DioException(
            requestOptions: native.requestOptions,
            type: DioExceptionType.connectionError,
            response: native,
            stackTrace: stack,
          ),
        );
      final error = _expectError(
        await ValueResponse('https://example.test', () => dio).call(),
        TonikErrorType.network,
        response: true,
      );
      _check(identical(error.response, native), 'transport response identity');
      _check(error.stackTrace.toString().isNotEmpty, 'transport stack');
    case 'synchronous-cancel':
      final native = _nativeResponse();
      final exceptionStack = StackTrace.fromString('exception cancel stack');
      final caughtStack = StackTrace.fromString('caught cancel stack');
      final exception = DioException(
        requestOptions: native.requestOptions,
        type: DioExceptionType.cancel,
        response: native,
        stackTrace: exceptionStack,
      );
      final error = _expectError(
        await ValueResponse(
          'https://example.test',
          () => _SynchronousThrowDio(exception, caughtStack),
        ).call(),
        TonikErrorType.cancelled,
        response: true,
      );
      _check(identical(error.response, native), 'sync cancel response identity');
      _check(
        error.stackTrace.toString() == caughtStack.toString(),
        'sync cancel caught stack',
      );
    case 'synchronous-transport':
      final native = _nativeResponse();
      final exceptionStack = StackTrace.fromString('exception network stack');
      final caughtStack = StackTrace.fromString('caught network stack');
      final exception = DioException(
        requestOptions: native.requestOptions,
        type: DioExceptionType.connectionError,
        response: native,
        stackTrace: exceptionStack,
      );
      final error = _expectError(
        await ValueResponse(
          'https://example.test',
          () => _SynchronousThrowDio(exception, caughtStack),
        ).call(),
        TonikErrorType.network,
        response: true,
      );
      _check(
        identical(error.response, native),
        'sync transport response identity',
      );
      _check(
        error.stackTrace.toString() == caughtStack.toString(),
        'sync transport caught stack',
      );
    case 'future-error-stack':
      final native = _nativeResponse();
      final exceptionStack = StackTrace.fromString('exception future stack');
      final deliveredStack = StackTrace.fromString('delivered future stack');
      final exception = DioException(
        requestOptions: native.requestOptions,
        type: DioExceptionType.connectionError,
        response: native,
        stackTrace: exceptionStack,
      );
      final error = _expectError(
        await ValueResponse(
          'https://example.test',
          () => _FutureErrorDio(exception, deliveredStack),
        ).call(),
        TonikErrorType.network,
        response: true,
      );
      _check(identical(error.response, native), 'future response identity');
      _check(
        error.stackTrace.toString() == deliveredStack.toString(),
        'future delivered stack',
      );
    case 'unexpected-transport':
      final dio = Dio()..httpClientAdapter = _ThrowAdapter(StateError('boom'));
      _expectError(
        await ValueResponse('https://example.test', () => dio).call(),
        TonikErrorType.network,
        response: false,
      );
    case 'decoding':
      final error = _expectError(
        await ValueResponse('https://example.test', _invalidJsonDio).call(),
        TonikErrorType.decoding,
        response: true,
      );
      _check(error.response is Response<List<int>>, 'completed Dio response');
      _check(error.stackTrace.toString().isNotEmpty, 'decoding stack');
    case 'success':
      final result = await ValueResponse(
        'https://example.test',
        _successDio,
      ).call();
      _check(result is TonikSuccess<String, Response<Object?>>, 'success type');
      final success = result as TonikSuccess<String, Response<Object?>>;
      _check(success.value == 'ok', 'decoded value');
      _check(success.response is Response<List<int>>, 'native response');
    case 'void':
      final result = await VoidResponse(
        'https://example.test',
        _emptyDio,
      ).call();
      _check(result is TonikSuccess<void, Response<Object?>>, 'void result');
    case 'never':
      final result = await NeverResponse(
        'https://example.test',
        _nullDio,
      ).call();
      _expectError(result, TonikErrorType.decoding, response: true);
    case 'nullable-never':
      final result = await NullableNeverResponse(
        'https://example.test',
        _nullDio,
      ).call();
      _check(
        result is TonikSuccess<Never?, Response<Object?>>,
        'Never? result',
      );
    case 'none':
      final result = await NoDeclaredResponse(
        'https://example.test',
        _emptyDio,
      ).call();
      _check(result is TonikSuccess<void, Response<Object?>>, 'none result');
    case 'multipart':
      final adapter = _CaptureAdapter();
      final dio = Dio()..httpClientAdapter = adapter;
      final result = await MultipartRequest(
        'https://example.test',
        () => dio,
      ).call(body: const UploadBody(label: 'runtime'));
      _check(
        result is TonikSuccess<void, Response<Object?>>,
        'multipart result',
      );
      _check(adapter.data is FormData, 'multipart data');
    case 'synchronous-forwarding':
      final delegate = Dio()..httpClientAdapter = _BodyAdapter('"ok"');
      final dio = _RequestUriProbeDio(delegate);
      final future = ValueResponse('https://example.test', () => dio).call();
      _check(
        dio.entered,
        'Dio request preparation was scheduled instead of forwarded',
      );
      await future;
    default:
      throw StateError('unknown scenario $scenario');
  }
}

Dio _unusedDio() => throw StateError('client accessor must not run');

Dio _successDio() => Dio()..httpClientAdapter = _BodyAdapter('"ok"');
Dio _invalidJsonDio() => Dio()..httpClientAdapter = _BodyAdapter('not-json');
Dio _nullDio() => Dio()..httpClientAdapter = _BodyAdapter('null');
Dio _emptyDio() => Dio()..httpClientAdapter = _BodyAdapter('');

Response<Object?> _nativeResponse() => Response<Object?>(
  requestOptions: RequestOptions(path: '/'),
  statusCode: 503,
);

TonikError<dynamic, dynamic> _expectError(
  Object result,
  TonikErrorType type, {
  required bool response,
}) {
  _check(
    result is TonikError<dynamic, dynamic>,
    'expected TonikError: $result',
  );
  final error = result as TonikError<dynamic, dynamic>;
  _check(error.type == type, 'expected $type, got ${error.type}');
  _check((error.response != null) == response, 'response boundary');
  return error;
}

void _check(bool condition, String message) {
  if (!condition) throw StateError(message);
}

class _BodyAdapter implements HttpClientAdapter {
  _BodyAdapter(this.body);
  final String body;

  @override
  Future<ResponseBody> fetch(
    RequestOptions options,
    Stream<Uint8List>? requestStream,
    Future<void>? cancelFuture,
  ) async => ResponseBody.fromBytes(
    utf8.encode(body),
    200,
    headers: {
      'content-type': ['application/json'],
    },
  );

  @override
  void close({bool force = false}) {}
}

class _ThrowAdapter implements HttpClientAdapter {
  _ThrowAdapter(this.error);
  final Object error;

  @override
  Future<ResponseBody> fetch(
    RequestOptions options,
    Stream<Uint8List>? requestStream,
    Future<void>? cancelFuture,
  ) async => throw error;

  @override
  void close({bool force = false}) {}
}

class _CancelAdapter implements HttpClientAdapter {
  _CancelAdapter(this.cancellation);
  final TonikCancellation cancellation;

  @override
  Future<ResponseBody> fetch(
    RequestOptions options,
    Stream<Uint8List>? requestStream,
    Future<void>? cancelFuture,
  ) async {
    scheduleMicrotask(() => cancellation.cancel('during'));
    await cancelFuture;
    return ResponseBody.fromString('"late"', 200);
  }

  @override
  void close({bool force = false}) {}
}

class _CaptureAdapter implements HttpClientAdapter {
  bool entered = false;
  Object? data;

  @override
  Future<ResponseBody> fetch(
    RequestOptions options,
    Stream<Uint8List>? requestStream,
    Future<void>? cancelFuture,
  ) async {
    entered = true;
    data = options.data;
    return ResponseBody.fromString('', 200);
  }

  @override
  void close({bool force = false}) {}
}

class _SynchronousThrowDio implements Dio {
  _SynchronousThrowDio(this.error, this.stackTrace);

  final Object error;
  final StackTrace stackTrace;

  @override
  Future<Response<T>> requestUri<T>(
    Uri uri, {
    Object? data,
    CancelToken? cancelToken,
    Options? options,
    ProgressCallback? onSendProgress,
    ProgressCallback? onReceiveProgress,
  }) => Error.throwWithStackTrace(error, stackTrace);

  @override
  dynamic noSuchMethod(Invocation invocation) => super.noSuchMethod(invocation);
}

class _FutureErrorDio implements Dio {
  _FutureErrorDio(this.error, this.stackTrace);

  final Object error;
  final StackTrace stackTrace;

  @override
  Future<Response<T>> requestUri<T>(
    Uri uri, {
    Object? data,
    CancelToken? cancelToken,
    Options? options,
    ProgressCallback? onSendProgress,
    ProgressCallback? onReceiveProgress,
  }) => Future<Response<T>>.error(error, stackTrace);

  @override
  dynamic noSuchMethod(Invocation invocation) => super.noSuchMethod(invocation);
}

final class _DioEncodingOperation extends DioOperation<String> {
  _DioEncodingOperation() : super('https://example.test', _unusedDio);

  Future<TonikResult<String, Response<Object?>>> run(StackTrace stack) =>
      execute(
        prepare: () => Error.throwWithStackTrace(StateError('encoding'), stack),
        decode: (_) => 'unused',
      );
}

class _RequestUriProbeDio implements Dio {
  _RequestUriProbeDio(this._delegate);

  final Dio _delegate;
  bool entered = false;

  @override
  Future<Response<T>> requestUri<T>(
    Uri uri, {
    Object? data,
    CancelToken? cancelToken,
    Options? options,
    ProgressCallback? onSendProgress,
    ProgressCallback? onReceiveProgress,
  }) {
    entered = true;
    return _delegate.requestUri<T>(
      uri,
      data: data,
      cancelToken: cancelToken,
      options: options,
      onSendProgress: onSendProgress,
      onReceiveProgress: onReceiveProgress,
    );
  }

  @override
  dynamic noSuchMethod(Invocation invocation) => super.noSuchMethod(invocation);
}
