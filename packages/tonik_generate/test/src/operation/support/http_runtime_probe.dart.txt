import 'dart:async';
import 'dart:convert';

import 'package:http/http.dart';
import 'package:http_runtime_api/src/model/upload_body.dart';
import 'package:http_runtime_api/src/operation/http_operation.dart';
import 'package:http_runtime_api/src/operation/multipart_request.dart'
    as generated;
import 'package:http_runtime_api/src/operation/never_response.dart';
import 'package:http_runtime_api/src/operation/no_declared_response.dart';
import 'package:http_runtime_api/src/operation/nullable_never_response.dart';
import 'package:http_runtime_api/src/operation/value_response.dart';
import 'package:http_runtime_api/src/operation/void_response.dart';
import 'package:tonik_util/tonik_util.dart';

Future<void> main(List<String> arguments) async {
  final scenario = arguments.single;
  switch (scenario) {
    case 'encoding':
      final stack = StackTrace.fromString('http encoding stack');
      final error = _expectError(
        await _HttpEncodingOperation().run(stack),
        TonikErrorType.encoding,
        response: false,
      );
      _check(error.stackTrace.toString() == stack.toString(), 'encoding stack');
    case 'uri-encoding':
      _expectError(
        await ValueResponse('http://[', _unusedClient).call(),
        TonikErrorType.encoding,
        response: false,
      );
    case 'pre-cancel':
      final cancellation = TonikCancellation()..cancel('before');
      final result = await ValueResponse(
        'https://example.test',
        () => _Client(_success),
      ).call(cancellation: cancellation);
      final error = _expectError(
        result,
        TonikErrorType.cancelled,
        response: false,
      );
      _check(error.error is RequestAbortedException, 'abort error');
      _check(error.stackTrace.toString().isNotEmpty, 'StackTrace.current');
    case 'accessor':
      final stack = StackTrace.fromString('http accessor stack');
      Client accessor() =>
          Error.throwWithStackTrace(StateError('closed'), stack);
      final error = _expectError(
        await ValueResponse('https://example.test', accessor).call(),
        TonikErrorType.other,
        response: false,
      );
      _check(error.stackTrace.toString() == stack.toString(), 'accessor stack');
    case 'abort-cancelled':
      final cancellation = TonikCancellation();
      final client = _Client((request) async {
        cancellation.cancel('during');
        throw RequestAbortedException(request.url);
      });
      _expectError(
        await ValueResponse(
          'https://example.test',
          () => client,
        ).call(cancellation: cancellation),
        TonikErrorType.cancelled,
        response: false,
      );
    case 'abort-network':
      final client = _Client(
        (request) async => throw RequestAbortedException(request.url),
      );
      _expectError(
        await ValueResponse('https://example.test', () => client).call(),
        TonikErrorType.network,
        response: false,
      );
    case 'client-exception':
      final stack = StackTrace.fromString('client exception stack');
      final client = _Client(
        (_) => Future<StreamedResponse>.error(
          ClientException('network'),
          stack,
        ),
      );
      final error = _expectError(
        await ValueResponse('https://example.test', () => client).call(),
        TonikErrorType.network,
        response: false,
      );
      _check(error.stackTrace.toString() == stack.toString(), 'client stack');
    case 'timeout':
      final client = _Client(
        (_) => Future<StreamedResponse>.error(TimeoutException('late')),
      );
      _expectError(
        await ValueResponse('https://example.test', () => client).call(),
        TonikErrorType.network,
        response: false,
      );
    case 'unexpected-send':
      final client = _Client((_) async => throw StateError('unexpected'));
      _expectError(
        await ValueResponse('https://example.test', () => client).call(),
        TonikErrorType.other,
        response: false,
      );
    case 'buffering':
      final client = _Client(
        (_) async => StreamedResponse(
          Stream<List<int>>.error(StateError('buffer')),
          200,
        ),
      );
      _expectError(
        await ValueResponse('https://example.test', () => client).call(),
        TonikErrorType.network,
        response: false,
      );
    case 'decoding':
      final error = _expectError(
        await ValueResponse(
          'https://example.test',
          () => _Client(_invalidJson),
        ).call(),
        TonikErrorType.decoding,
        response: true,
      );
      _check(error.response is Response, 'completed HTTP response');
      _check(error.stackTrace.toString().isNotEmpty, 'decoding stack');
    case 'success':
      final result = await ValueResponse(
        'https://example.test',
        () => _Client(_success),
      ).call();
      _check(result is TonikSuccess<String, Response>, 'success type');
      final success = result as TonikSuccess<String, Response>;
      _check(success.value == 'ok', 'decoded value');
      _check(success.response.statusCode == 200, 'native response');
    case 'void':
      final result = await VoidResponse(
        'https://example.test',
        () => _Client(_empty),
      ).call();
      _check(result is TonikSuccess<void, Response>, 'void result');
    case 'never':
      final result = await NeverResponse(
        'https://example.test',
        () => _Client(_null),
      ).call();
      _expectError(result, TonikErrorType.decoding, response: true);
    case 'nullable-never':
      final result = await NullableNeverResponse(
        'https://example.test',
        () => _Client(_null),
      ).call();
      _check(result is TonikSuccess<Never?, Response>, 'Never? result');
    case 'none':
      final result = await NoDeclaredResponse(
        'https://example.test',
        () => _Client(_empty),
      ).call();
      _check(result is TonikSuccess<void, Response>, 'none result');
    case 'multipart':
      BaseRequest? captured;
      final client = _Client((request) async {
        captured = request;
        return _empty(request);
      });
      final result = await generated.MultipartRequest(
        'https://example.test',
        () => client,
      ).call(body: const UploadBody(label: 'runtime'));
      _check(result is TonikSuccess<void, Response>, 'multipart result');
      _check(captured is MultipartRequest, 'multipart construction');
      final multipart = captured! as MultipartRequest;
      _check(
        multipart.fields.containsValue('runtime') || multipart.files.isNotEmpty,
        'multipart payload',
      );
    case 'synchronous-forwarding':
      var entered = false;
      final client = _Client((request) {
        entered = true;
        return _success(request);
      });
      final future = ValueResponse(
        'https://example.test',
        () => client,
      ).call();
      _check(entered, 'HTTP send was scheduled instead of forwarded');
      await future;
    default:
      throw StateError('unknown scenario $scenario');
  }
}

Client _unusedClient() => throw StateError('client accessor must not run');

Future<StreamedResponse> _success(BaseRequest request) async =>
    StreamedResponse(
      Stream.value(utf8.encode('"ok"')),
      200,
      request: request,
      headers: {'content-type': 'application/json'},
    );

Future<StreamedResponse> _invalidJson(BaseRequest request) async =>
    StreamedResponse(
      Stream.value(utf8.encode('not-json')),
      200,
      request: request,
      headers: {'content-type': 'application/json'},
    );

Future<StreamedResponse> _null(BaseRequest request) async => StreamedResponse(
  Stream.value(utf8.encode('null')),
  200,
  request: request,
  headers: {'content-type': 'application/json'},
);

Future<StreamedResponse> _empty(BaseRequest request) async =>
    StreamedResponse(Stream.empty(), 200, request: request);

TonikError<dynamic, dynamic> _expectError(
  Object result,
  TonikErrorType type, {
  required bool response,
}) {
  _check(
    result is TonikError<dynamic, dynamic>,
    'expected TonikError: $result',
  );
  final error = result as TonikError<dynamic, dynamic>;
  _check(error.type == type, 'expected $type, got ${error.type}');
  _check((error.response != null) == response, 'response boundary');
  return error;
}

void _check(bool condition, String message) {
  if (!condition) throw StateError(message);
}

class _Client extends BaseClient {
  _Client(this.callback);

  final Future<StreamedResponse> Function(BaseRequest) callback;

  @override
  Future<StreamedResponse> send(BaseRequest request) => callback(request);
}

final class _HttpEncodingOperation extends HttpOperation<String> {
  _HttpEncodingOperation()
      : super('https://example.test', _unusedClient);

  Future<TonikResult<String, Response>> run(StackTrace stack) => execute(
        prepare: () => Error.throwWithStackTrace(StateError('encoding'), stack),
        decode: (_) => 'unused',
      );
}
